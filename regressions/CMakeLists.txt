include(ProcessorCount)

find_package(Threads REQUIRED)

ProcessorCount(CK_REGRESSION_CORES_DEFAULT)
set(CK_REGRESSION_CORES "${CK_REGRESSION_CORES_DEFAULT}"
    CACHE STRING "Number of cores used by regression programs"
)
if(NOT CK_REGRESSION_CORES MATCHES "^[1-9][0-9]*$")
    message(FATAL_ERROR "CK_REGRESSION_CORES must be a positive integer")
endif()

math(EXPR CK_REGRESSION_HALF_CORES "${CK_REGRESSION_CORES} / 2")
if(CK_REGRESSION_HALF_CORES LESS 1)
    set(CK_REGRESSION_HALF_CORES 1)
endif()

add_library(ck_regression_support INTERFACE)
target_compile_definitions(ck_regression_support
    INTERFACE _GNU_SOURCE CORES=${CK_REGRESSION_CORES}
)
if(MSVC_VERSION)
    add_library(ck_pthread_compat STATIC win32/pthread.c)
    target_include_directories(ck_pthread_compat PUBLIC win32)
    target_link_libraries(ck_regression_support INTERFACE ck_pthread_compat)
else()
    target_link_libraries(ck_regression_support INTERFACE Threads::Threads)
endif()
if(UNIX)
    target_link_libraries(ck_regression_support INTERFACE m)
endif()

if(TARGET ck_static)
    set(CK_REGRESSION_LIBRARY ck_static)
else()
    set(CK_REGRESSION_LIBRARY ck_shared)
endif()

function(_ck_add_regression_executable kind name)
    if(kind STREQUAL "validate")
        cmake_parse_arguments(args "DISABLED" "LIBRARY;OUT_TARGET" "SOURCES;DEFINITIONS;ARGS"
            ${ARGN}
        )
    else()
        cmake_parse_arguments(args "" "LIBRARY;OUT_TARGET" "SOURCES;DEFINITIONS" ${ARGN})
    endif()
    if(NOT DEFINED args_SOURCES)
        set(args_SOURCES "${name}.c")
    endif()
    if(NOT DEFINED args_LIBRARY)
        set(args_LIBRARY ${CK_REGRESSION_LIBRARY})
    endif()

    get_filename_component(suite_dir "${CMAKE_CURRENT_SOURCE_DIR}" DIRECTORY)
    get_filename_component(suite "${suite_dir}" NAME)
    string(REPLACE "." "_" identifier "${name}")
    set(target "${suite}_${kind}_${identifier}")

    add_executable(${target} ${args_SOURCES})
    target_link_libraries(${target} PRIVATE ck_regression_support ${args_LIBRARY})
    target_compile_definitions(${target} PRIVATE ${args_DEFINITIONS})
    set_target_properties(${target} PROPERTIES OUTPUT_NAME "${name}")
    add_dependencies(regressions ${target})

    if(kind STREQUAL "validate")
        set(test "${suite}/${name}")
        add_test(NAME "${test}" COMMAND ${target} ${args_ARGS})
        if(args_DISABLED)
            set_tests_properties("${test}" PROPERTIES DISABLED TRUE)
        endif()
    endif()
    if(DEFINED args_OUT_TARGET)
        set(${args_OUT_TARGET} "${target}" PARENT_SCOPE)
    endif()
endfunction()

# Forward without adding a scope so OUT_TARGET reaches the leaf.
macro(ck_add_regression name)
    _ck_add_regression_executable(validate "${name}" ${ARGN})
endmacro()

macro(ck_add_benchmark name)
    _ck_add_regression_executable(benchmark "${name}" ${ARGN})
endmacro()

# Keep assertions enabled in the regression subtree in optimized builds.
foreach(config RELEASE RELWITHDEBINFO MINSIZEREL)
    string(REGEX REPLACE
        "[ ]*[/-]DNDEBUG[ ]*" " "
        CMAKE_C_FLAGS_${config} "${CMAKE_C_FLAGS_${config}}"
    )
endforeach()
